# 앱이 어떤 API 서버를 어떤 경로로 부르는지 알아내는 방법

## 어셈블리 저장소 추출
```
# 작업 폴더 생성 후 저장소 라이브러리만 추출
mkdir -p <경로 생성>
unzip -o <split_abi>.apk 'lib/<abi>/libassembly-store.so' -d native_libs

# 헤더로 형식과 어셈블리 개수를 확인(굳이 안해도 됨)
python3 -c "import struct,sys;d=open(sys.argv[1],'rb').read();i=d.find(b'XABA');print('magic offset',i);print('version 0x%08x entries %d index_entries %d index_size %d'%struct.unpack_from('<5I',d,i)[1:])" native_libs/lib/<abi>/libassembly-store.so

# LZ4 해제 모듈 (apt 대신 pip 로, 시스템 파이썬 안 건드림)
python3 -m pip install --user lz4

# 어셈블리 추출
python3 extract_assembly_store.py native_libs/lib/<abi>/libassembly-store.so assemblies
```

## 공격 표면 추출
```
# 꺼낸 게 정상 .NET 어셈블리인지 확인
file assemblies/*.dll

# 2바이트 문자열을 attack_surface.py 로 범주별 정리
strings -el assemblies/*.dll | python3 attack_surface.py -
```

## .NET MAUI 어셈블리 디컴파일
```
# 어셈블리의 .NET 버전 확인
strings -el <assemblies_dir>/System.Private.CoreLib.dll | grep -iE "^[0-9]+\.[0-9]+\.[0-9]+" | head

# 맞는 .NET SDK 준비
curl -sSL https://dot.net/v1/dotnet-install.sh -o /tmp/dotnet-install.sh
bash /tmp/dotnet-install.sh --channel <major.minor> --install-dir "$HOME/.dotnet<major>"
export DOTNET_ROOT="$HOME/.dotnet<major>"; export PATH="$HOME/.dotnet<major>:$PATH"
dotnet --version

# ILSpy 설치
dotnet tool install -g ilspycmd --version <ilspy_version>
export PATH="$PATH:$HOME/.dotnet/tools"
ilspycmd --version

# 디컴파일
ilspycmd -p -o <out_dir>/<app_name> <assemblies_dir>/<app_name>.dll
find <out_dir> -name '*.cs' | wc -l

# 공격 표면 찾기
cd <out_dir>
grep -rln -iE "<pattern1>|<pattern2>" .
```
